Feature	Library
Every	strategy	card	below	calls	these	shared,	OHLC-only	helper	functions	by	name	instead	of	
redefining	them.	Implement	once	in	M3	(feature
layer);	every	strategy	in	M4	is	then	a	pure	function	of	these	outputs	
plus	the	current	regime	label.
atr(bars,	i,	n=14)
Wilder	ATR.	TR	=	max(H-L,	|H-C[-1]|,	|L-C[-1]|);	ATR	=	RMA(TR,	n).
ema(bars,	i,	n)
Exponential	moving	average,	alpha	=	2/(n+1).
sma(values)
Simple	average	of	a	list	of	values.
rsi(bars,	i,	n=14)
Wilder	RSI	from	average	gain	/	average	loss	over	n	bars.
adx(bars,	i,	n=14)
Wilder	ADX:	+DM/-DM	->	DI+/DI-	->	DX	->	RMA(DX,	n).	Trend	strength,	0-
100.
efficiency_ratio(bars,	i,	n)
Kaufman	ER	=	|C	-	C[-n]|	/	sum(|C[k]-C[k-1]|).	Straightness	of	the	move,	0-1.
ols_r2(bars,	i,	n)
R-squared	of	an	OLS	regression	of	Close	on	bar	index	over	the	last	n	bars.
variance_ratio(bars,	i,	q)
Var(q-bar	returns)	/	(q	*	Var(1-bar	returns)).	>1	trending,	<1	mean-
reverting.
hurst(bars,	i,	window)
Hurst	exponent	via	DFA	on	log	returns.	<0.5	mean-reverting,	>0.5	trending.
swing_high	/ 	swing_low(bars,	i)
Fractal	pivot:	local	extreme	vs.	k	bars	on	each	side	(k=2	default).
market_state(bars,	i)
'bull'	(HH/HL)	|	'bear'	(LH/LL)	|	'chop',	from	the	tracked	pivot	sequence.
structure_is(pattern,	bars,	i)
True	if	the	last	two	tracked	swings	match	a	named	pattern,	e.g.	'HH_HL'.
bollinger_bands(bars,	i,	n,	k)
SMA(n)	+/-	k*stdev(n);	returns	(lower,	mid,	upper).
bollinger_bandwidth(bars,	i,	n,	k)
(upper	-	lower)	/	mid,	from	bollinger_bands().
bbwp(bars,	i,	lookback)
Percentile	rank	of	current	Bollinger	Bandwidth	over	the	lookback	window.
keltner	/ 	squeeze_active(bars,	i)
EMA(20)	+/-	1.5*ATR;	squeeze	=	Bollinger	Bands	inside	the	Keltner	Channel.
zscore(bars,	i,	n)
(Close	-	SMA(n))	/	stdev(n).
anchored_twap(bars,	i)
Mean(Close)	since	the	session/anchor	open	--	the	volume-free	VWAP
substitute.
supertrend(bars,	i,	m,	n)
ATR-trailing	trend	line;	returns	{value,	direction}.
adf_pvalue(residual)	/ 	ols_residual(a,	b,	i)
Cointegration	residual	and	its	ADF	stationarity	p-value,	for	pairs.
true_range(bars,	i)
Current-bar	TR	(see	atr	above),	unsmoothed.
Order	/ 	PairOrder	/ 	Ladder
Common	return	objects:	entry,	stop,	targets	(price,	%,	action),	and	optional
kill/trail	rules.

Trend	Pullback	to	Dynamic	Support

GATE	(ALL	MUST	HOLD)
state	=	HH/HL	(bull)	or	LH/LL	(bear)
ADX(14)	>	25
ER(20)	>	0.3

TRIGGER
Closed	bar	touches	EMA20	±	0.25×ATR;	reversal	bar	(close	in	top
40%	of	range,	low	>	prior	swing	low);	RSI(14)	>	40.

intraday

def	signal_A1(bars,	i):
				state	= 	market_state(bars,	i)																	#	'bull'	|	'bear'	|	'chop'
				if	state	= = 	'chop':
								return	None
				if	not	(adx(bars,	i,	14)	>	25	and	efficiency_ratio(bars,	i,	20)	>	0.3
												and	ols_r2(bars,	i,	50)	>	0.6):
								return	None
				atr_v,	ema20	= 	atr(bars,	i,	14),	ema(bars,	i,	20)
				c	= 	bars[i]
				touched		= 	abs(c.close	-	ema20)	<= 	0.25	*	atr_v
				reversal	= 	close_position(c)	>= 	0.40	and	c.low	>	swing_low(bars,	i)
				if	not	(touched	and	reversal	and	rsi(bars,	i,	14)	>	40):
								return	None
				direction	= 	'long'	if	state	= = 	'bull'	else	'short'
				entry	= 	c.high	+	TICK	if	direction	= = 	'long'	else	c.low	-	TICK
				raw_stop	= 	(min(swing_low(bars,	i),	c.low)	-	0.25	*	atr_v	if	direction	= = 	'long'
																else	max(swing_high(bars,	i),	c.high)	+	0.25	*	atr_v)
				if	abs(entry	-	raw_stop)	>	2	*	atr_v:
								return	None																																	#	stop	too	wide	->	skip
				r	= 	abs(entry	-	raw_stop)
				tp1	= 	entry	+	r	if	direction	= = 	'long'	else	entry	-	r
				tp2	= 	prior_swing(bars,	i,	direction)	or	(entry	+	2.5*atr_v	if	direction= = 'long'
																																															else	entry	-	2.5*atr_v)
				return	Order('A1',	direction,	'stop',	entry,	raw_stop,
																	targets= [(tp1,	0.5,	'stop_to_breakeven'),
																										(tp2,	0.3,	None),
																										(None,	0.2,	'trail_chandelier_22_3ATR')],
																	time_stop_bars= 10,	flat_at_session_close= True)

def	signal_A2(bars,	i):
				if	market_state(bars,	i)	= = 	'chop':
								return	None
				piv	= 	last_pivot_high(bars,	i)
				atr_v	= 	atr(bars,	i,	14)
				c	= 	bars[i]
				broke	= 	c.close	>	piv	+	0.1*atr_v	and	close_position(c)	>= 	0.70
				if	not	(broke	and	rising(efficiency_ratio,	bars,	i,	20)):
								return	None
				if	opposing_htf_level_within(bars,	i,	1*atr_v):
								return	None
				entry	= 	c.close																																		#	or	retest	limit	at	`piv`
				stop		= 	piv	-	0.5*atr_v
				if	abs(entry	-	stop)	<	1.25*atr_v:
								stop	= 	entry	-	1.25*atr_v
				swing_h	= 	prior_swing_height(bars,	i)
				tp	= 	entry	+	(swing_h	or	2*abs(entry-stop))
				order	= 	Order('A2',	'long',	'market',	entry,	stop,
																		targets= [(tp,	1.0,	None)])
				order.kill_if	= 	lambda	b,	j:	b[j].close	<	piv					#	within	3	bars	of	entry
				return	order

def	signal_A3(bars,	i):
				if	market_state(bars,	i)	= = 	'chop':
								return	None
				atr_v	= 	atr(bars,	i,	14)
				ribbon	= 	ribbon_spread(bars,	i)	/	atr_v
				if	not	(ribbon	<	0.5	and	bbwp(bars,	i,	100)	<	25):
								return	None
				flag_hi,	flag_lo	= 	flag_range(bars,	i)
				c	= 	bars[i]
				tr_v	= 	true_range(bars,	i)
				direction	= 	trend_direction(bars,	i)
				breakout	= 	((c.close	>	flag_hi)	if	direction	= = 	'long'	else	(c.close	<	flag_lo))																and	tr_v	>	1.3*atr_v
				if	not	breakout:
								return	None
				entry	= 	c.close
				stop		= 	(flag_lo	-	0.25*atr_v)	if	direction	= = 	'long'	else	(flag_hi	+	0.25*atr_v)
				flag_h	= 	flag_hi	-	flag_lo
				r	= 	abs(entry	-	stop)
				tp1	= 	entry	+	1.5*r	if	direction= = 'long'	else	entry	-	1.5*r
				tp2	= 	entry	+	flag_h	if	direction= = 'long'	else	entry	-	flag_h
				return	Order('A3',	direction,	'stop',	entry,	stop,
																	targets= [(tp1,	0.5,	None),	(tp2,	0.5,	None)])

def	signal_B1(bars,	i,	n= 20):
				if	atr_percentile_rank(bars,	i,	250)	<= 	30:
								return	None
				hh,	ll	= 	max(b.high	for	b	in	bars[i-n:i]),	min(b.low	for	b	in	bars[i-n:i])
				c	= 	bars[i]
				if	efficiency_ratio(bars,	i,	20)	<= 	0.25:
								return	None
				if	c.close	>	hh:
								direction,	level	= 	'long',	hh
				elif	c.close	<	ll:
								direction,	level	= 	'short',	ll
				else:
								return	None
				if	tagged_twice_this_week(bars,	i,	level):
								return	None																																			#	likely	a	liquidity	grab,	not	a	break
				atr_v	= 	atr(bars,	i,	14)
				entry	= 	level	+	TICK	if	direction	= = 	'long'	else	level	-	TICK
				stop_a	= 	entry	-	1.5*atr_v	if	direction= = 'long'	else	entry	+	1.5*atr_v

				stop	= 	max(stop_a,	stop_b)	if	direction= = 'long'	else	min(stop_a,	stop_b)

																	targets= [(None,	1.0,	'trail_chandelier_3ATR')],
